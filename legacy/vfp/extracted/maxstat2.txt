PROCEDURE creaexcel
#INCLUDE excel9.h
#DEFINE false .F.
#DEFINE true .T.

#DEFINE RPT_FONTNAME "Arial"
#DEFINE RPT_FONTSIZE 8

PUBLIC loexcel, lcolderror, lcrange, lnsheets, lncounter

WAIT WINDOW NOWAIT "Starting Excel..."
lcolderror = ON("ERROR")
ON ERROR loexcel = .NULL.
loexcel = GETOBJECT(, "Excel.Application")
ON ERROR &lcolderror

IF ISNULL(loexcel)
   loexcel = CREATEOBJECT( "Excel.Application" )
ENDIF

WITH loexcel

   .workbooks.ADD
   .displayalerts = false
   lnsheets = .sheets.COUNT
   FOR lncounter = 1 TO lnsheets - 1
      .sheets(1).DELETE
   ENDFOR
   .sheets(1).SELECT
   .sheets(1).NAME = "Max Order"

   .activewindow.displaygridlines = true
   .RANGE("A1").VALUE = "RAM FOOD PRODUCTS, INC."
   .RANGE("A1:D1").mergecells=.t.
   .RANGE("A2").VALUE = "MAXORDER RESULT" 
   .RANGE("A2:D2").mergecells=.t.
   .RANGE("A3").VALUE = "DATE: " + upper(MDY(date()))
   .RANGE("A3:D3").mergecells=.t.
   m = exc_font("A1:A4",10,RPT_FONTNAME,xlLeft,.T.)

   .RANGE("A6").VALUE="P.O. #"
   .RANGE("B6").VALUE="P.O. DATE"
   .RANGE("C6").VALUE="CUSTKEY"
   .RANGE("D6").VALUE="CUSTOMER NAME"
   .RANGE("E6").VALUE="PRODUCT"
   .RANGE("F6").VALUE="ORDERED CS"
   .RANGE("G6").VALUE="ORDERED PC"
   .RANGE("H6").VALUE="NEW PC"
   .RANGE("I6").VALUE="NEW CS"
   .RANGE("J6").VALUE="REMARKS"
   m = exc_font("A6:J6",9,RPT_FONTNAME,xlCenter,.T.)

   lnrow=7
   lncol=1
   sele dmaxstatx1
   mrec=reccount()
   go top
   if mrec<>0
      do while !eof()
         wait wind "Populating Cells..."+str(mrec,12) nowait
         mrec=mrec-1
          mpodate1 = podate
          mln = alltrim(str(lnrow))
          
         .range("A"+alltrim(str(lnrow))).value = pono &&ponum
         .range("B"+alltrim(str(lnrow))).value = dtoc(mpodate1)
         .range("C"+alltrim(str(lnrow))).value = allt(custkey)
         .range("D"+alltrim(str(lnrow))).value = allt(cusname)
         .range("E"+alltrim(str(lnrow))).value = allt(cprodno)
         .range("F"+alltrim(str(lnrow))).value = origcs
         .range("G"+alltrim(str(lnrow))).value = origpc
         .range("H"+alltrim(str(lnrow))).value = newcs
         .range("I"+alltrim(str(lnrow))).value = newpc
         .Range("F"+mln+":I"+mln).HorizontalAlignment = xlRight
*!*	         .Range("F"+mln+":I"+mln).HorizontalAlignment = xlRight     
         .range("J"+alltrim(str(lnrow))).value = allt(remarks)
         
         lnrow = lnrow + 1
         
         sele dmaxstatx1
         skip
      enddo   

   endif
   
   xlnrow=alltr(str(lnrow-1))
   
   .RANGE("A1:A"+xlnrow).columnwidth=10
   .RANGE("A7:J"+xlnrow).FONT.NAME = RPT_FONTNAME
   .RANGE("A7:J"+xlnrow).FONT.SIZE = 8
   .RANGE("A1:J"+xlnrow).EntireColumn.AutoFit
   
   m = exc_box("A6:J"+xlnrow,xlNone,xlContinuous,xlThin,xlAutomatic,xldiagonalDown,xlDiagonalup,xlEdgeleft,xlEdgetop,xlEdgebottom,xlEdgeright,xlInsideVertical,xlInsideHorizontal)
   WITH .activesheet.pagesetup
      .ORIENTATION = xlportrait
      .PAPERSIZE = xlPaperLetter
      .printtitlerows = "$1:$7"
*      .printtitlecolumns = "$A:$C"
      .LeftFooter = "Page &P of &N"
      .RightFooter= dirloc()
   ENDWITH
   .activeworkbook.SAVEAS(mexcel)
ENDWITH

WAIT CLEAR
MESSAGEBOX(mexcel,64,"CREATED")
loexcel.VISIBLE = .T.
RELEASE loexcel
RETURN
ENDPROC

PROCEDURE Click
*!*11/23/16	CCUETO 
sele dmaxstatx1
go top
if reccount() <> 0
   mexcdir = "C:\XLSFILES"
   if .not. directory(mexcdir)
      md &mexcdir
   endif
   mctr = 1
   mexcel = addbs(mexcdir)  + "MAXORDER RESULT " + upper(mdy(date())) + "("+alltr(str(mctr)) +").XLS"   
   mcheckmaxfile = .t.
   do while mcheckmaxfile
      if file(mexcel)
         mctr = mctr + 1
         mexcel = addbs(mexcdir)  + "MAXORDER RESULT " + upper(mdy(date())) + "("+alltr(str(mctr)) +").XLS"
      else
         mcheckmaxfile = .f.
      endif
   enddo
   thisform.creaexcel
else
   messagebox("No record to process",64,"Warning")
*!*	   MESSAGEBOX(mexcel,64,"CREATED")
   return
endif
ENDPROC
