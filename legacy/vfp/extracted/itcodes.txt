PROCEDURE dchecklast
param xnum
n1 = val(substr(xnum,1,1))
n2 = val(substr(xnum,2,1))
n3 = val(substr(xnum,3,1))
n4 = val(substr(xnum,4,1))
n5 = val(substr(xnum,5,1)) && will be used in validation

mres1 = n1 + n3
mres2 = n2 + n4
mres3 = mres1 * mres2
mres4 = iif(mres3<10,"0"+str(mres3,1),left(allt(str(mres3)),2))
mres5 = abs(val(substr(mres4,1,1))-val(substr(mres4,2,1)))
if n5 <> mres5
   xret = xret + "1"
endif

ENDPROC
PROCEDURE Init
param xtitle, xtcode
this.caption = xtitle
thisform.dtcode = xtcode
thisform.dretval = ""
thisform.refresh
ENDPROC
PROCEDURE Unload
return thisform.dretval
ENDPROC

--- METHODS ---
PROCEDURE Click
mcode = allt(thisform.txtCode.value)
mcode12 = substr(mcode,1,12)
mcodebr = substr(mcode,1,2)
m = openfile("itcodes")
mused = .f.
if .not. used("sysparam")
	m = openfile("sysparam")
	mused = .t.
endif
sele sysparam
go top
mbcode = alltr(sysparam.bcode)
if mbcode = mcodebr
   sele itcodes
   set order to ncode12
   seek mcode12
   if .not. found()
      mtcode = substr(mcode,13,3)
      if thisform.dtcode = mtcode
         xret = ""
         m = thisform.dchecklast(substr(mcode,1,5))
         m = thisform.dchecklast(substr(mcode,7,5))
         if len(xret) = 0
            thisform.dretval = mcode
         endif
      endif
   endif
endif
m = closedbf("itcodes")
if mused
	m = closedbf("sysparam")
endif
if empty(thisform.dretval) 
   m = messageb("Entered Code is INVALID. Please check",64,"Code Check")
else
   release thisform
endif
ENDPROC
