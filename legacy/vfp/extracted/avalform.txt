--- PROPERTIES ---

--- METHODS ---
PROCEDURE definelist
thisform.list1.rowsource=thisform.dalias
thisform.list1.rowsourcetype=6
thisform.list1.columncount=2

thisform.list1.columnwidths="50,380"
thisform.label1.width=50
thisform.label2.width=385
thisform.label1.left=13
thisform.label2.left=65
thisform.label1.caption="Code"
thisform.label2.caption="Description"
if type("thisform.dsearchkey")="L"
   thisform.dsearchkey="Code"
else   
   if thisform.dsearchkey<>"Code"
      thisform.list1.columnwidths="380,50"
      thisform.label1.left=400
      thisform.label2.left=13
   endif
endif   
sele(thisform.dalias)

do case
   case upper(thisform.dalias)="PROD4WIN"
      thisform.dretfield="cprodno"
      thisform.list1.columncount=5
*      thisform.label1.caption="prodno"
*      thisform.label2.caption="product description"
      if thisform.dsearchkey="Code"
         set order to cprodno
         thisform.list1.columnwidths="50,270,35,35,35"
         thisform.list1.rowsource=thisform.dalias+".cprodno,proddesc,pieces,qtyperpc,um"
      else
         set order to proddesc
         thisform.list1.columnwidths="270,35,35,35,50"
         thisform.list1.rowsource=thisform.dalias+".proddesc,pieces,qtyperpc,um,cprodno"
      endif
   case upper(thisform.dalias)="CUST4WIN"
      thisform.list1.columncount=3

      thisform.dretfield="ckey"
      if thisform.dsearchkey="Code"
         thisform.list1.columnwidths="50,330,50"
         set order to ckey
         thisform.list1.rowsource=thisform.dalias+".ckey,cusname,csman"
      else
         thisform.list1.columnwidths="330,50,50"

         set order to cusname
         thisform.list1.rowsource=thisform.dalias+".cusname,csman,ckey"
      endif
   case upper(thisform.dalias)="SUP4WIN"
      thisform.dretfield="csupno"
      if thisform.dsearchkey="Code"
         set order to csupno
         thisform.list1.rowsource=thisform.dalias+".csupno,supname"
      else
         set order to supname
         thisform.list1.rowsource=thisform.dalias+".supname,csupno"
      endif
   case upper(thisform.dalias)="DEAL4MAS"
      thisform.dretfield="dealcode"
      if thisform.dsearchkey="Code"
         set order to dealcode
         thisform.list1.rowsource=thisform.dalias+".dealcode,dealdesc"
      else
         set order to dealdesc
         thisform.list1.rowsource=thisform.dalias+".dealdesc,dealcode"
      endif
   case upper(thisform.dalias)="SMAN4WIN"
      thisform.dretfield="csman"
      if thisform.dsearchkey="Code"
         set order to csman
         set filter to .not. empty(subd)
         go top
         thisform.list1.rowsource=thisform.dalias+".csman,name"
      else
         set order to name
         set filter to .not. empty(subd)
         go top
         thisform.list1.rowsource=thisform.dalias+".name,csman"
      endif
   case upper(thisform.dalias)="BRANCHW"
      thisform.dretfield="whseno"
      if thisform.dsearchkey="Code"
         set order to whseno
         thisform.list1.rowsource=thisform.dalias+".whseno,branch"
      else
         set order to branch
         thisform.list1.rowsource=thisform.dalias+".branch,whseno"
      endif
   case upper(thisform.dalias)="TERMS"
      thisform.dretfield="term"
      if thisform.dsearchkey="Code"
         set order to term
         thisform.list1.rowsource=thisform.dalias+".term,desc"
      else
         set order to desc
         thisform.list1.rowsource=thisform.dalias+".desc,term"
      endif
   case upper(thisform.dalias)="CHANNEL"
      thisform.dretfield="channel"
      if thisform.dsearchkey="Code"
         set order to channel
         thisform.list1.rowsource=thisform.dalias+".channel,chname"
      else
         set order to chname
         thisform.list1.rowsource=thisform.dalias+".chname,channel"
      endif
   case upper(thisform.dalias)="VAN"
      thisform.dretfield="vanno"
      if thisform.dsearchkey="Code"
         set order to vanno
         thisform.list1.rowsource=thisform.dalias+".vanno,vandesc"
      else
         set order to vandesc
         thisform.list1.rowsource=thisform.dalias+".vandesc,vanno"
      endif
   case upper(thisform.dalias)="DOCNUM"
      thisform.dretfield="doctype"
      if thisform.dsearchkey="Code"
         set order to doctype
         thisform.list1.rowsource=thisform.dalias+".docid,doctype"
      else
         set order to doctype
         thisform.list1.rowsource=thisform.dalias+".doctype,docid"
      endif
   case upper(thisform.dalias)="ZONEMAST"
      thisform.dretfield="czone"
      if thisform.dsearchkey="Code"
         set order to czone
         thisform.list1.rowsource=thisform.dalias+".czone,cdesc"
      else
         set order to cdesc
         thisform.list1.rowsource=thisform.dalias+".cdesc,czone"
      endif
   case upper(thisform.dalias)="TEMPSMAN"
      midai1 = newname("xai1")
      thisform.dretfield="csman"
      if thisform.dsearchkey="Code"
         index on csman to &midai1
         go top
         thisform.list1.rowsource=thisform.dalias+".csman,name"
      else
         index on name to &midai1
         go top
         thisform.list1.rowsource=thisform.dalias+".name,csman"
      endif
      case upper(thisform.dalias)="TEMPSMAN1"
      midaix = newname("xaix")
      thisform.dretfield="csman"
      if thisform.dsearchkey="Code"
         index on csman to &midaix
         go top
         thisform.list1.rowsource=thisform.dalias+".csman,name"
      else
         index on name to &midaix
         go top
         thisform.list1.rowsource=thisform.dalias+".name,csman"
      endif
endcase
go top 

mdata = thisform.dalias+"."+thisform.dretfield
if type("&mdata")="C"
   thisform.dreturnvalue=""
else
   thisform.dreturnvalue=0
endif      
     
thisform.refresh
ENDPROC
PROCEDURE Unload
m = delfile("dsman")
return thisform.dreturnvalue
ENDPROC
PROCEDURE Init
param dcaption,dalias
thisform.caption = "SELECT "+dcaption
*thisform.drecordsource=drecordsource
*thisform.dretfield=dretfield
thisform.dalias=dalias
sele (thisform.dalias)
go top
thisform.definelist
thisform.dreturnvalue=""
thisform.dsearchkey="Code"
ENDPROC
