PROCEDURE recalc
wait wind "Recalculating Inventory for accurate available and allocated quantities please wait..." nowait
tempavail = newname("DBF")
thisdate=sysparam.transdate     
sele avail4
copy to &tempavail for .not. empty(cprodno) .and. class = 1
use
use &tempavail in 0 alias tempavail exclusive
sele tempavail
delete all for allt(cprodno)="0"
pack
if type("dusrfdlose") = "U"
   alter table &tempavail add dusrfdlose n(9,2)
endif
do recalc16 with "tempavail",thisdate     && update qtycs from imtr
do recalc1b with "tempavail",thisdate    && update allocated from oowk

sele prod4win
set order to cprodno
sele tempavail
set rela to cprodno into prod4win
** ai 07/12/2006 - update supplier no of eif
repl all supplier with prod4win.supplier for cprodno = prod4win.cprodno
go top
do while .not. eof()
   mpieces=prod4win.pieces
   wait window "Updating TOHCS, TOHPC of prodno "+cprodno nowait
   mtohcs = (begcs+rrcs+othincs - (salescs + othiscs)) - dusrfdcs
   mtohpc = (begpc+rrpc+othinpc - (salespc + othispc)) - dusrfdpc
   mtohloose = (begloose + inloose - outloose) - dusrfdlose
   repl tohcs with mtohcs, tohpc with mtohpc
   if type("TOHLOOSE") = "N"
       repl tohloose with mtohloose
   endif
   mloose=0
   mcases=0
   
   mqtypc = qtyloose(tohcs,tohpc,tohloose,prod4win.pieces,prod4win.innerqty,4)
   mcases = qtyloose(tohcs,tohpc,tohloose,prod4win.pieces,prod4win.innerqty,1)   
   mpcs = qtyloose(tohcs,tohpc,tohloose,prod4win.pieces,prod4win.innerqty,2)   
   mloose = qtyloose(tohcs,tohpc,tohloose,prod4win.pieces,prod4win.innerqty,3)
   repl tohcs with mcases,tohpc with mpcs,tohloose with mloose
   repl qtycs with tohcs - alloccs,qtypc with tohpc - allocpc
   m=depcs("qtycs","qtypc",mpieces)
   skip
enddo   
sele tempavail


ENDPROC

      endif
      mmofr = eom(gomonth(mmofr,1))
   enddo
endif

if mmaxproceed
   mF = maxqty - mtotqty
   if mF <= 0 && total orders is now equal to maxorder
      issugmax  = .t.
      sele tempowkd
      dele for docno = mdocno and cprodno = mcprodno
      appe blank
      repl docno with mdocno, supplier with msupplier, class with mclass, prodno with val(mcprodno), cprodno with mcprodno, qtycs with 0, qtypc with 0

      seek mcprodno

      else
         remcs = mqtycs
