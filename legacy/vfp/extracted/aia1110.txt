PROCEDURE Init
param mdoctypex, mproc
thisform.dproc = mproc
do case
   case mdoctypex = "SO"
      thisform.chk_SO.visible = .t.
   otherwise
      thisform.chk_SO.visible = .f.
endcase

tempsman = newname("dtmpsman")
midsman = newname("xsman")
sele sman4win
copy to &tempsman for .not. empty(subd)
use &tempsman alias tempsman in 0 excl
sele tempsman
index on csman to &midsman
go top
if mproc = "ADD"
   thisform.dsman = tempsman.salesman
   thisform.dsname = tempsman.name
   thisform.drgwno = 0
   thisform.drgwdate = {}
   thisform.drgwamt = 0
   thisform.djobwhse = 0
   thisform.djobckey = space(5)
else
   thisform.dsman = xsman
   thisform.dsname = xsname
   if xsocheck
      thisform.chk_SO.visible = .t.
      thisform.chk_SO.value = 1
      thisform.chk_SO.click
      thisform.drgwno = xrgwno
      thisform.drgwdate = xrgwdate
      thisform.drgwamt = xrgwamt
      thisform.djobwhse = xjobwhse
      thisform.djobckey = xjobckey
      thisform.txtJobname.value = xjobname
   else
      thisform.chk_SO.visible = .f.
      thisform.drgwno = 0
      thisform.drgwdate = {}
      thisform.drgwamt = 0
      thisform.djobwhse = 0
      thisform.djobckey = space(5)
   endif
   thisform.cmdOk.enabled = .t.
endif

*m = openfile("sman4win")
*mtempsman = newname("ds1")
*sele sman4win
*copy to &mtempsman for .not. empty(subd)
*use &mtempsman alias mtempsman in 0 excl
*sele mtempsman

*thisform.cboSman.rowsourcetype = 6
*thisform.cboSman.rowsource = "mtempsman.salesman,name"
*m = openfile("sman4win")
thisform.refresh


ENDPROC
PROCEDURE Unload
*m = closedbf("sman4win")
m = delfile("tempsman")
xretval = str(thisform.dsman,4)+thisform.dsname+str(thisform.drgwno,8)+dtoc(thisform.drgwdate)+str(thisform.drgwamt,14,2)+str(thisform.djobwhse,2)+thisform.djobckey+thisform.dcommand
return xretval

ENDPROC
