PROCEDURE Init
param dcaption,dalias
*thisform.lblCaption.caption = "SELECT "+dcaption
thisform.caption = "SELECT "+dcaption
*thisform.drecordsource=drecordsource
*thisform.dretfield=dretfield
thisform.dalias=dalias
sele (thisform.dalias)
go top
thisform.definelist
thisform.dreturnvalue=""
thisform.dsearchkey="Code"
ENDPROC
PROCEDURE Unload
if upper(thisform.dalias)="TMPTRUCK"
   erase &a1
   erase &a2
endif
return thisform.dreturnvalue
ENDPROC
